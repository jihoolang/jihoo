// Comments, literals and meaningful newlines survive formatting.

#[derive(Show)]
struct Point {
    x: i64
    y: i64
}

fn main() {
    var value = 1
    let pointer = &value
    unsafe { *pointer += 1 }
    let copy = unsafe { *(pointer as *i64) }
    let negated = -(value + 1)
    let total = 1 + // keep this comment
        2 * 3
    let text = "가  // not a comment /* either */ \x41"
    /* outer /* nested */
   comment body stays here */
    println(text)
    println(copy + negated)
    println(total)
    println(value - -1)
    let pair = (1, 2)
    println(pair.0)
    match value { -2..=9 => println(value), _ => {} }
    println(match value {
        3 => 30
        _ => 0
    })
    let Point { x, .. } = Point { x: 4,
        y: 5 }
    println(x)
}
